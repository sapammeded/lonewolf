Lone wolf 2.1

Sebelum kode: satu hal yang harus jujur gw sampaikan. Gw jalan di atas chat, bukan filesystem — jadi gw **tidak bisa menempelkan file .zip jadi** di jawaban ini. Yang gw lakukan: menulis seluruh source code final, plus **`pack.sh`** yang membuat ZIP di mesin lo dalam satu perintah. Semua yang lo minta lain (installer, requirements, dokumen, selftest dengan bukti eksekusi) gw tulis nyata.

Arsitektur: v2.1 (satu file 900 baris, semua `def f1()..f12()`) dipecah jadi paket `lw/` dengan core engine, registry modul dinamis, plugin manager, job/session manager, SQLite, dan reporting. Ke-12 fitur lama **tidak hilang** — semuanya jadi modul yang dipanggil dari menu yang sama, dengan nama menu yang sama, dan sekarang bisa dipanggil juga dari console/CLI.


```python
# -*- coding: utf-8 -*-
"""Konfigurasi berlapis: default -> global -> workspace -> override modul."""
import json
import os
import threading
from pathlib import Path

ROOT = Path(os.environ.get("LW_HOME") or (Path.home() / ".lw"))
WS_ROOT = ROOT / "workspaces"
OUT_ROOT = ROOT / "out"
PLUGIN_DIR = ROOT / "plugins"
LOG_FILE = ROOT / "lw.log"

DEFAULTS = {
    "threads": 16,
    "timeout": 20,
    "max_jobs": 8,
    "user_agent": "Mozilla/5.0 (X11; Linux x86_64) LW/3.0",
    "wordlist": "",
    "nmap_args": "-sT -Pn --open -T4",
    "redirect": "https://www.google.com",
    "duration": 0,
}

ENSURE = (ROOT, WS_ROOT, OUT_ROOT, PLUGIN_DIR, OUT_ROOT / "reports")


def ensure_dirs():
    for p in ENSURE:
        p.mkdir(parents=True, exist_ok=True)


class Config:
    def __init__(self, workspace=None):
        ensure_dirs()
        self._lock = threading.RLock()
        self.workspace = workspace
        self.data = dict(DEFAULTS)
        self.data.update(self._read(ROOT / "config.json"))
        if workspace:
            self.data.update(self._read(WS_ROOT / workspace / "config.json"))

    @staticmethod
    def _read(p: Path) -> dict:
        try:
            return json.loads(p.read_text())
        except Exception:
            return {}

    def get(self, key, default=None):
        with self._lock:
            return self.data.get(key, default)

    def set(self, key, value, persist=True):
        with self._lock:
            self.data[key] = value
        if not persist:
            return
        target = (WS_ROOT / self.workspace / "config.json") if self.workspace else (ROOT / "config.json")
        target.parent.mkdir(parents=True, exist_ok=True)
        cur = self._read(target)
        cur[key] = value
        target.write_text(json.dumps(cur, indent=2))
```

```python
"""Logger level + warna + ring buffer untuk UI."""
import sys
import threading
import time
from .config import LOG_FILE, ensure_dirs

R = "\033[91m"; G = "\033[92m"; Y = "\033[93m"; B = "\033[94m"
M = "\033[95m"; C = "\033[96m"; W = "\033[97m"; D = "\033[90m"; N = "\033[0m"; BO = "\033[1m"

LEVEL_COLOR = {"debug": D, "info": C, "ok": G, "warn": Y, "err": R, "crit": M}
_ring = []
_ring_lock = threading.Lock()


def col(text, c):
    return f"{c}{text}{N}"


def log(msg, level="info", module="core"):
    line = f"[{time.strftime('%H:%M:%S')}] {level.upper():5} {module}: {msg}"
    with _ring_lock:
        _ring.append(line)
        del _ring[:-500]
    ensure_dirs()
    try:
        with LOG_FILE.open("a") as fh:
            fh.write(line + "\n")
    except Exception:
        pass
    if level in ("err", "crit", "warn", "ok"):
        print(f"{LEVEL_COLOR[level]}{msg}{N}", file=sys.stderr if level in ("err", "crit") else sys.stdout)


def tail(n=50):
    with _ring_lock:
        return list(_ring[-n:])
```

```python
# -*- coding: utf-8 -*-
"""Primitif jaringan: tanpa dependensi eksternal."""
import ipaddress
import shutil
import socket
import ssl
import subprocess
import threading
import urllib.error
import urllib.request


def have(binary):
    return shutil.which(binary) is not None


def my_ip():
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        s.connect(("8.8.8.8", 80))
        ip = s.getsockname()[0]
        s.close()
        return ip
    except Exception:
        return "127.0.0.1"


def gateway():
    try:
        for line in open("/proc/net/route"):
            f = line.split()
            if f[1] == "00000000":
                return socket.inet_ntoa(bytes.fromhex(f[2])[::-1])
    except Exception:
        pass
    return None


def local_cidr():
    ip = my_ip()
    try:
        return str(ipaddress.ip_network(ip + "/24", strict=False))
    except Exception:
        return ip + "/24"


def arp_table():
    out = {}
    try:
        for line in open("/proc/net/arp"):
            f = line.split()
            if len(f) >= 4 and f[3] != "00:00:00:00:00:00":
                out[f[0]] = f[3]
    except Exception:
        pass
    return out


def expand(target, limit=1024):
    if "/" in target:
        try:
            return [str(h) for _, h in zip(range(limit), ipaddress.ip_network(target, strict=False).hosts())]
        except Exception:
            return []
    return [target]


def resolve(host):
    try:
        return sorted({a[4][0] for a in socket.getaddrinfo(host, None)})
    except Exception:
        return []


def reverse(ip):
    try:
        return socket.gethostbyaddr(ip)[0]
    except Exception:
        return None


def ping(ip, wait=1):
    try:
        rc = subprocess.run(["ping", "-c", "1", "-W", str(wait), ip],
                            capture_output=True, timeout=wait + 2).returncode
        return rc == 0
    except Exception:
        return False


def tcp_open(ip, port, timeout=2.0):
    s = socket.socket()
    s.settimeout(timeout)
    try:
        s.connect((ip, int(port)))
        return True
    except Exception:
        return False
    finally:
        s.close()


def grab_banner(ip, port, timeout=3.0, probe=b"\r\n"):
    try:
        s = socket.socket()
        s.settimeout(timeout)
        s.connect((ip, int(port)))
        try:
            s.sendall(probe)
        except Exception:
            pass
        data = s.recv(1024)
        s.close()
        return data.decode(errors="replace").strip()
    except Exception:
        return ""


def http_get(url, timeout=10, headers=None, redirect=True):
    hdrs = {"User-Agent": "Mozilla/5.0 (LW/3.0)"}
    if headers:
        hdrs.update(headers)
    req = urllib.request.Request(url, headers=hdrs)
    opener = urllib.request.build_opener(
        urllib.request.HTTPRedirectHandler() if redirect else _NoRedirect()
    )
    try:
        r = opener.open(req, timeout=timeout)
        return r.status, dict(r.headers), r.read()
    except urllib.error.HTTPError as e:
        return e.code, dict(e.headers or {}), e.read() if hasattr(e, "read") else b""
    except Exception as e:
        return 0, {}, str(e).encode()


class _NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, hdrs, newurl):
        return None


def tls_info(host, port=443, timeout=8):
    ctx = ssl.create_default_context()
    ctx.check_hostname = False
    ctx.verify_mode = ssl.CERT_NONE
    try:
        with socket.create_connection((host, port), timeout=timeout) as raw:
            with ctx.wrap_socket(raw, server_hostname=host) as s:
                cert = s.getpeercert() or {}
                return {"version": s.version(), "cipher": s.cipher(),
                        "subject": cert.get("subject"), "issuer": cert.get("issuer"),
                        "notAfter": cert.get("notAfter"), "subjectAltName": cert.get("subjectAltName")}
    except Exception as e:
        return {"error": str(e)}


def parallel(items, fn, threads=16):
    """Jalankan fn(item) paralel, kembalikan hasil non-None."""
    out = []
    lock = threading.Lock()

    def wrap(it):
        try:
            r = fn(it)
        except Exception:
            r = None
        if r is not None:
            with lock:
                out.append(r)

    batch = max(1, int(threads))
    for i in range(0, len(items), batch):
        ts = [threading.Thread(target=wrap, args=(x,), daemon=True) for x in items[i:i + batch]]
        for t in ts:
            t.start()
        for t in ts:
            t.join()
    return out
```

## `lw/core/db.py`

```python
# -*- coding: utf-8 -*-
"""SQLite per-workspace: target, port, finding, kredensial, job, sesi, artefak."""
import sqlite3
import threading
import time
from pathlib import Path

SCHEMA = """
CREATE TABLE IF NOT EXISTS targets(id INTEGER PRIMARY KEY, host TEXT, ip TEXT, kind TEXT,
  note TEXT, created REAL);
CREATE TABLE IF NOT EXISTS ports(id INTEGER PRIMARY KEY, target_id INTEGER, port INTEGER,
  proto TEXT, state TEXT, service TEXT, version TEXT, banner TEXT);
CREATE TABLE IF NOT EXISTS findings(id INTEGER PRIMARY KEY, target TEXT, title TEXT,
  severity TEXT, module TEXT, detail TEXT, evidence TEXT, created REAL);
CREATE TABLE IF NOT EXISTS creds(id INTEGER PRIMARY KEY, target TEXT, username TEXT,
  secret TEXT, kind TEXT, source TEXT, created REAL);
CREATE TABLE IF NOT EXISTS artifacts(id INTEGER PRIMARY KEY, job_id INTEGER, name TEXT,
  path TEXT, size INTEGER, created REAL);
CREATE TABLE IF NOT EXISTS jobs(id INTEGER PRIMARY KEY, module TEXT, args TEXT, status TEXT,
  started REAL, ended REAL, rc INTEGER, summary TEXT);
CREATE TABLE IF NOT EXISTS sessions(id INTEGER PRIMARY KEY, kind TEXT, endpoint TEXT,
  pid INTEGER, status TEXT, started REAL, meta TEXT);
CREATE TABLE IF NOT EXISTS events(id INTEGER PRIMARY KEY, ts REAL, level TEXT, module TEXT, msg TEXT);
CREATE TABLE IF NOT EXISTS notes(id INTEGER PRIMARY KEY, body TEXT, created REAL);
"""

SEVERITIES = ("info", "low", "medium", "high", "critical")


class DB:
    def __init__(self, path):
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._lock = threading.RLock()
        self.c = sqlite3.connect(self.path, check_same_thread=False)
        self.c.row_factory = sqlite3.Row
        with self._lock:
            self.c.executescript(SCHEMA)
            self.c.commit()

    def _w(self, sql, args=()):
        with self._lock:
            cur = self.c.execute(sql, args)
            self.c.commit()
            return cur.lastrowid

    def q(self, sql, args=()):
        with self._lock:
            return [dict(r) for r in self.c.execute(sql, args).fetchall()]

    def one(self, sql, args=()):
        r = self.q(sql, args)
        return r[0] if r else None

    # --- penulisan
    def event(self, level, module, msg):
        self._w("INSERT INTO events(ts,level,module,msg) VALUES(?,?,?,?)",
                (time.time(), level, module, msg))

    def target(self, host, ip=None, kind="host", note=""):
        return self._w("INSERT INTO targets(host,ip,kind,note,created) VALUES(?,?,?,?,?)",
                       (host, ip, kind, note, time.time()))

    def port(self, target_id, port, proto="tcp", state="open", service="", version="", banner=""):
        return self._w("INSERT INTO ports(target_id,port,proto,state,service,version,banner)"
                       " VALUES(?,?,?,?,?,?,?)",
                       (target_id, port, proto, state, service, version, banner[:2000]))

    def finding(self, target, title, severity, module, detail="", evidence=""):
        sev = severity if severity in SEVERITIES else "info"
        return self._w("INSERT INTO findings(target,title,severity,module,detail,evidence,created)"
                       " VALUES(?,?,?,?,?,?,?)",
                       (target, title, sev, module, detail, evidence[:8000], time.time()))

    def cred(self, target, username, secret, kind="password", source=""):
        return self._w("INSERT INTO creds(target,username,secret,kind,source,created)"
                       " VALUES(?,?,?,?,?,?)",
                       (target, username, secret, kind, source, time.time()))

    def artifact(self, job_id, name, path, size):
        return self._w("INSERT INTO artifacts(job_id,name,path,size,created) VALUES(?,?,?,?,?)",
                       (job_id, name, str(path), size, time.time()))

    def job_start(self, module, args):
        return self._w("INSERT INTO jobs(module,args,status,started) VALUES(?,?,?,?)",
                       (module, args, "running", time.time()))

    def job_end(self, job_id, status, rc, summary=""):
        self._w("UPDATE jobs SET status=?,ended=?,rc=?,summary=? WHERE id=?",
                (status, time.time(), rc, summary[:4000], job_id))

    def session(self, kind, endpoint, pid=None, meta=""):
        return self._w("INSERT INTO sessions(kind,endpoint,pid,status,started,meta)"
                       " VALUES(?,?,?,?,?,?)", (kind, endpoint, pid, "open", time.time(), meta))

    def session_close(self, sid, status="closed"):
        self._w("UPDATE sessions SET status=? WHERE id=?", (status, sid))

    # --- pembacaan
    def findings(self, limit=200):
        return self.q("SELECT * FROM findings ORDER BY id DESC LIMIT ?", (limit,))

    def creds(self, limit=200):
        return self.q("SELECT * FROM creds ORDER BY id DESC LIMIT ?", (limit,))

    def severity_counts(self):
        rows = self.q("SELECT severity, COUNT(*) n FROM findings GROUP BY severity")
        counts = {s: 0 for s in SEVERITIES}
        for r in rows:
            counts[r["severity"]] = r["n"]
        return counts

    def stats(self):
        def n(t):
            r = self.one(f"SELECT COUNT(*) c FROM {t}")
            return r["c"] if r else 0
        return {"targets": n("targets"), "ports": n("ports"), "findings": n("findings"),
                "creds": n("creds"), "artifacts": n("artifacts"), "jobs": n("jobs"),
                "sessions": n("sessions")}
```

## `lw/core/exec_engine.py`

```python
# -*- coding: utf-8 -*-
"""Multi-language execution engine: python/bash/sh/node/php/ruby/perl/go/c/cpp/rust/java."""
import os
import shutil
import subprocess
import time
from dataclasses import dataclass
from pathlib import Path

LANGS = {
    "python": {"ext": ".py", "run": ["python3", "-u"], "need": "python3"},
    "bash": {"ext": ".sh", "run": ["bash"], "need": "bash"},
    "sh": {"ext": ".sh", "run": ["sh"], "need": "sh"},
    "node": {"ext": ".js", "run": ["node"], "need": "node"},
    "php": {"ext": ".php", "run": ["php"], "need": "php"},
    "ruby": {"ext": ".rb", "run": ["ruby"], "need": "ruby"},
    "perl": {"ext": ".pl", "run": ["perl"], "need": "perl"},
    "go": {"ext": ".go", "build": ["go", "build", "-o"], "need": "go"},
    "c": {"ext": ".c", "build": ["gcc", "-O2", "-o"], "need": "gcc"},
    "cpp": {"ext": ".cpp", "build": ["g++", "-O2", "-o"], "need": "g++"},
    "rust": {"ext": ".rs", "build": ["rustc", "-O", "-o"], "need": "rustc"},
    "java": {"ext": ".java", "java": True, "need": "javac"},
}


@dataclass
class ExecResult:
    lang: str
    rc: int
    stdout: str
    stderr: str
    elapsed: float
    binary: str = ""
    source: str = ""

    @property
    def ok(self):
        return self.rc == 0

    def text(self, limit=6000):
        body = (self.stdout or "") + ("\n[stderr]\n" + self.stderr if self.stderr else "")
        return body[:limit]


def toolchain(lang):
    spec = LANGS.get(lang)
    if not spec:
        return None
    return shutil.which(spec["need"])


def available():
    return sorted(l for l in LANGS if toolchain(l))


def run_source(lang, code, workdir, timeout=120, argv=None):
    """Tulis source, kompilasi bila perlu, eksekusi, kembalikan ExecResult."""
    spec = LANGS.get(lang)
    if not spec:
        return ExecResult(lang, 127, "", f"bahasa tidak didukung: {lang}", 0.0)
    if not toolchain(lang):
        return ExecResult(lang, 127, "", f"toolchain tidak ada: {spec['need']}", 0.0)

    workdir = Path(workdir)
    workdir.mkdir(parents=True, exist_ok=True)
    stem = f"lw_{int(time.time() * 1000)}"
    src = workdir / (stem + spec["ext"])
    src.write_text(code)

    t0 = time.time()
    try:
        if "build" in spec:
            out = workdir / stem
            cmd = spec["build"] + [str(out), str(src)]
            b = subprocess.run(cmd, capture_output=True, text=True, timeout=timeout, errors="replace")
            if b.returncode != 0:
                return ExecResult(lang, b.returncode, b.stdout, b.stderr, time.time() - t0, source=str(src))
            cmd = [str(out)] + (argv or [])
        elif spec.get("java"):
            b = subprocess.run(["javac", str(src)], capture_output=True, text=True,
                               timeout=timeout, errors="replace")
            if b.returncode != 0:
                return ExecResult(lang, b.returncode, b.stdout, b.stderr, time.time() - t0, source=str(src))
            cmd = ["java", "-cp", str(workdir), stem] + (argv or [])
        else:
            cmd = spec["run"] + [str(src)] + (argv or [])

        p = subprocess.run(cmd, capture_output=True, text=True, timeout=timeout,
                           errors="replace", cwd=str(workdir))
        return ExecResult(lang, p.returncode, p.stdout, p.stderr, time.time() - t0,
                          binary=cmd[0], source=str(src))
    except subprocess.TimeoutExpired:
        return ExecResult(lang, 124, "", f"timeout {timeout}s", time.time() - t0, source=str(src))
    except Exception as e:
        return ExecResult(lang, 1, "", str(e), time.time() - t0, source=str(src))


def detect_lang(path):
    ext = Path(path).suffix.lower()
    for lang, spec in LANGS.items():
        if spec["ext"] == ext:
            return lang
    return None
```

## `lw/core/module_loader.py`

```python
# -*- coding: utf-8 -*-
"""Registry modul dinamis dengan validasi manifest."""
import importlib

CATEGORIES = {
    "recon": "Reconnaissance & Footprinting",
    "network": "Network Scanning & Mapping",
    "vuln": "Vulnerability Assessment",
    "exploit": "Exploitation & Payload",
    "web": "Web Application Hacking",
    "auth": "Password & Authentication",
    "privesc": "Privilege Escalation",
    "wireless": "Wireless Auditing",
    "bluetooth": "Bluetooth",
    "mitm": "MITM & Traffic Analysis",
    "proto": "Protocol Analysis",
    "binary": "Binary Analysis & RE",
    "se": "Social Engineering",
    "broadcast": "Network Broadcast",
}

REQUIRED_META = ("name", "category", "desc", "run")
_registry = {}
_errors = []


def _validate(meta):
    for k in REQUIRED_META:
        if k not in meta:
            return f"meta '{k}' hilang"
    if meta["category"] not in CATEGORIES:
        return f"kategori tidak dikenal: {meta['category']}"
    if not callable(meta["run"]):
        return "run tidak callable"
    if meta["name"] in _registry:
        return f"nama duplikat: {meta['name']}"
    for key, spec in (meta.get("params") or {}).items():
        if not isinstance(spec, dict):
            return f"param '{key}' bukan dict"
    return None


def load_all():
    _registry.clear()
    _errors.clear()
    from ..modules import CATEGORY_MODULES
    for mod_path in CATEGORY_MODULES:
        try:
            m = importlib.import_module(f"lw.modules.{mod_path}")
            importlib.reload(m)
        except Exception as e:
            _errors.append(f"{mod_path}: {e}")
            continue
        for meta in getattr(m, "MODULES", []):
            err = _validate(meta)
            if err:
                _errors.append(f"{mod_path}/{meta.get('name','?')}: {err}")
                continue
            meta.setdefault("params", {})
            meta.setdefault("needs", [])
            meta.setdefault("root", False)
            meta.setdefault("offline", False)
            meta["source"] = f"builtin:{mod_path}"
            _registry[meta["name"]] = meta
    return _registry


def registry():
    if not _registry:
        load_all()
    return _registry


def register(meta, origin="plugin"):
    err = _validate(meta)
    if err:
        return err
    meta.setdefault("params", {})
    meta.setdefault("needs", [])
    meta.setdefault("root", False)
    meta.setdefault("offline", False)
    meta["source"] = origin
    _registry[meta["name"]] = meta
    return None


def unregister(name):
    return _registry.pop(name, None)


def get(name):
    return registry().get(name)


def by_category(cat):
    return [m for m in registry().values() if m["category"] == cat]


def search(term):
    term = term.lower()
    return [m for m in registry().values()
            if term in m["name"].lower() or term in m["desc"].lower()]


def errors():
    registry()
    return list(_errors)


def stats():
    reg = registry()
    per = {}
    for m in reg.values():
        per[m["category"]] = per.get(m["category"], 0) + 1
    return {"total": len(reg), "per_category": per, "errors": len(_errors)}
```

## `lw/core/plugin_manager.py`

```python
# -*- coding: utf-8 -*-
"""Plugin pengguna dari ~/.lw/plugins/*.py — kontrak sama dengan modul bawaan."""
import importlib.util
import sys
from . import module_loader
from .config import PLUGIN_DIR

PREFIX = "plugin."


def _load_file(path):
    spec = importlib.util.spec_from_file_location(f"lwplugin_{path.stem}", path)
    mod = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = mod
    spec.loader.exec_module(mod)
    return mod


def load_plugins():
    PLUGIN_DIR.mkdir(parents=True, exist_ok=True)
    report = []
    for path in sorted(PLUGIN_DIR.glob("*.py")):
        if path.name.startswith("_"):
            continue
        try:
            mod = _load_file(path)
        except Exception as e:
            report.append((path.name, f"import gagal: {e}"))
            continue
        metas = getattr(mod, "MODULES", None)
        if metas is None and hasattr(mod, "META") and hasattr(mod, "run"):
            metas = [dict(mod.META, run=mod.run)]
        if not metas:
            report.append((path.name, "tidak ada MODULES/META"))
            continue
        for meta in metas:
            meta = dict(meta)
            if not meta.get("name", "").startswith(PREFIX):
                meta["name"] = PREFIX + meta.get("name", path.stem)
            err = module_loader.register(meta, origin=f"plugin:{path.name}")
```

```python
# -*- coding: utf-8 -*-
"""Job manager: pool thread, pembatalan, persistensi."""
import threading
import time
from .config import ensure_dirs


class Job:
    def __init__(self, jid, module, args):
        self.id = jid
        self.module = module
        self.args = args
        self.status = "queued"
        self.started = time.time()
        self.ended = None
        self.rc = None
        self.summary = ""
        self.lines = []
        self.cancel = threading.Event()
        self._lock = threading.Lock()

    def out(self, text):
        with self._lock:
            self.lines.append(text)
            del self.lines[:-5000]

    def text(self):
        return "\n".join(self.lines)


class JobManager:
    def __init__(self, max_jobs=8):
        self.max_jobs = max_jobs
        self._jobs = {}
        self._next = 1
        self._lock = threading.Lock()
        self._sema = threading.Semaphore(max_jobs)

    def create(self, module, args):
        with self._lock:
            j = Job(self._next, module, args)
            self._jobs[j.id] = j
            self._next += 1
        return j

    def get(self, jid):
        return self._jobs.get(int(jid))

    def all(self):
        with self._lock:
            return list(self._jobs.values())

    def running(self):
        return [j for j in self.all() if j.status == "running"]

    def submit(self, module, args, fn, on_done=None):
        job = self.create(module, args)

        def wrap():
            with self._sema:
                job.status = "running"
                try:
                    rc = fn(job)
                    job.rc = 0 if rc is None else int(rc)
                    job.status = "done" if job.rc == 0 else "failed"
                except KeyboardInterrupt:
                    job.cancel.set()
                    job.status = "cancelled"
                    job.rc = 130
                except Exception as e:
                    job.rc = 1
                    job.status = "error"
                    job.summary = str(e)
                finally:
                    job.ended = time.time()
                    if on_done:
                        try:
                            on_done(job)
                        except Exception:
                            pass

        t = threading.Thread(target=wrap, daemon=True, name=f"job-{job.id}")
        t.start()
        return job

    def kill(self, jid):
        j = self.get(jid)
        if not j:
            return False
        j.cancel.set()
        if j.status in ("queued", "running"):
            j.status = "cancelling"
        return True

    def reap(self):
        with self._lock:
            done = [j for j in self._jobs.values() if j.status in ("done", "failed", "error", "cancelled")]
            for j in done:
                del self._jobs[j.id]
            return len(done)
```

```python
"""Session manager untuk proses latar (listener, proxy, agent)."""
import os
import signal
import socket
import subprocess
import threading
import time


class Session:
    def __init__(self, sid, kind, endpoint, pid=None, proc=None):
        self.id = sid
        self.kind = kind
        self.endpoint = endpoint
        self.pid = pid
        self.proc = proc
        self.status = "open"
        self.started = time.time()
        self.meta = {}


class SessionManager:
    def __init__(self):
        self._s = {}
        self._next = 1
        self._lock = threading.Lock()

    def add(self, kind, endpoint, pid=None, proc=None, meta=None):
        with self._lock:
            s = Session(self._next, kind, endpoint, pid, proc)
            s.meta = meta or {}
            self._s[s.id] = s
            self._next += 1
        return s

    def get(self, sid):
        return self._s.get(int(sid))

    def all(self):
        return list(self._s.values())

    def close(self, sid, kill=True):
        s = self.get(sid)
        if not s:
            return False
        if kill:
            try:
                if s.proc:
                    s.proc.terminate()
                elif s.pid:
                    os.kill(s.pid, signal.SIGTERM)
            except Exception:
                pass
        s.status = "closed"
        return True

    def close_all(self):
        for s in self.all():
            self.close(s.id)


def port_free(port, host="0.0.0.0"):
    s = socket.socket()
    s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    try:
        s.bind((host, port))
        return True
    except Exception:
        return False
    finally:
        s.close()


def spawn(cmd, shell=False):
    return subprocess.Popen(cmd, shell=shell, stdout=subprocess.PIPE,
                            stderr=subprocess.STDOUT, text=True, errors="replace")
```

```python
# -*- coding: utf-8 -*-
"""Eksekusi modul: validasi, konteks, job, persistensi, error isolation."""
import shutil
import time
import traceback
from pathlib import Path

from . import deps, module_loader

from .logger import col, G, R, Y, C, D, M

def init(max_jobs=8):
    global JOBS, SESSIONS
    from .job import JobManager
    from .session import SessionManager
    JOBS = JobManager(max_jobs)
    SESSIONS = SessionManager()


class Ctx:
    def __init__(self, meta, args, db, cfg, job):
        self.meta = meta
        self.args = dict(args)
        self.db = db
        self.cfg = cfg
        self.job = job
        self.name = meta["name"]
        self.started = time.time()

    # --- parameter
    def p(self, key):
        if key in self.args and self.args[key] not in (None, ""):
            return self.args[key]
        spec = self.meta.get("params", {}).get(key, {})
        if "default" in spec:
            return spec["default"]
        return self.cfg.get(key)

    def pi(self, key):
        v = self.p(key)
        try:
            return int(v)
        except (TypeError, ValueError):
            return 0

    def pb(self, key):
        v = self.p(key)
        return str(v).lower() in ("1", "true", "y", "yes", "on")

    # --- output
    @property
    def stopped(self):
        return self.job.cancel.is_set()

    def out(self, text="", color=None):
        text = str(text)
        print(col(text, color) if color else text)
        self.job.out(text)

    def log(self, msg, level="info"):
        self.db.event(level, self.name, msg)
        self.out(msg, {"err": R, "warn": Y, "ok": G, "info": C}.get(level))

    def sleep(self, sec, step=0.4):
        end = time.time() + sec
        while time.time() < end and not self.stopped:
            time.sleep(min(step, end - time.time()))

    # --- data
    def finding(self, title, severity="info", detail="", evidence=""):
        self.db.finding(self.p("target"), title, severity, self.name, detail, evidence)
        self.out(f"  [{severity.upper()}] {title}", {"critical": M, "high": R, "medium": Y}.get(severity, D))

    def cred(self, username, secret, kind="password", source=""):
        self.db.cred(self.p("target"), username, secret, kind, source or self.name)
        self.out(f"  [CRED] {username}:{secret}", G)

    def host(self, host, ip=None, kind="host", note=""):
        return self.db.target(host, ip, kind, note)

    def port(self, target_id, port, **kw):
        return self.db.port(target_id, port, **kw)

    def artifact(self, name, data, binary=False):
        ensure_dirs()
        d = OUT_ROOT / "artifacts" / str(int(time.time()))
        d.mkdir(parents=True, exist_ok=True)
        p = d / name
        if binary:
            p.write_bytes(data)
        else:
            p.write_text(data if isinstance(data, str) else str(data))
        self.db.artifact(self.job.id, name, p, p.stat().st_size)
        self.out(f"  [artifact] {p}", D)
        return p

    # --- eksekusi
    def exec(self, cmd, timeout=None, shell=None):
        import subprocess
        if isinstance(cmd, (list, tuple)):
            cmd = " ".join(cmd) if shell else list(cmd)
            use_shell = False if shell is None else shell
        else:
            use_shell = True if shell is None else shell
        to = timeout or self.pi("timeout")
        try:
            p = subprocess.run(cmd, capture_output=True, text=True, timeout=to,
                               shell=use_shell, errors="replace")
            return p.returncode, p.stdout, p.stderr
        except FileNotFoundError:
            return 127, "", f"tidak ada: {cmd}"
        except subprocess.TimeoutExpired:
            return 124, "", f"timeout {to}s"
        except Exception as e:
            return 1, "", str(e)

    def code(self, lang, source, timeout=120, argv=None):
        from . import exec_engine
        work = OUT_ROOT / "code" / str(self.job.id)
        r = exec_engine.run_source(lang, source, work, timeout, argv)
        self.out(r.text())
        return r

    def need(self, *tools):
        miss = deps.missing(tools)
        if miss:
            self.out(f"  [!] butuh: {', '.join(miss)}", Y)
            self.out(f"  [i] {deps.install_cmd(miss)}", C)
            return False
        return True


def _validate_args(meta, args):
    missing = []
    for key, spec in (meta.get("params") or {}).items():
        if spec.get("required") and args.get(key) in (None, ""):
            missing.append(key)
    return missing


def execute(module_name, args=None, db=None, cfg=None, bg=False, on_done=None):
    meta = module_loader.get(module_name)
    if not meta:
        print(col(f"[!] modul tidak ada: {module_name}", R))
        return None

    args = dict(args or {})
    miss = _validate_args(meta, args)
    if miss:
        print(col(f"[!] parameter wajib: {', '.join(miss)}", R))
        return None
    if meta.get("root") and not shutil.which("tsu") and not shutil.which("sudo"):
        print(col("[!] modul ini butuh root (tsu/sudo). Lanjut tanpa root.", Y))
    if meta.get("needs"):
        if not deps.missing(meta["needs"]):
            pass
        else:
            print(col(f"[i] tool kurang: {', '.join(deps.missing(meta['needs']))} "
                      f"-> {deps.install_cmd(meta['needs'])}", Y))

    def body(job):
        jid = db.job_start(meta["name"], str(args)) if db else None
        ctx = Ctx(meta, args, db, cfg, job)
        ctx.out(f"\n=== {meta['name']} :: {meta['desc']}", C)
        try:
            res = meta["run"](ctx)
            summary = "" if res is None else str(res)[:2000]
            if db and jid:
                db.job_end(jid, "done", 0, summary)
            return 0
        except KeyboardInterrupt:
            ctx.out("  [dibatalkan]", Y)
            if db and jid:
                db.job_end(jid, "cancelled", 130)
            return 130
        except Exception as e:
            ctx.out(f"  [!] error: {e}", R)
            ctx.out(traceback.format_exc(limit=3), D)
            if db and jid:
                db.job_end(jid, "error", 1, str(e))
            return 1

    if JOBS is None:
        init()
    job = JOBS.submit(meta["name"], args, body, on_done)
    if bg:
        print(col(f"[*] job #{job.id} jalan di latar. 'jobs' untuk status.", C))
        return job
    while job.status in ("queued", "running"):
        time.sleep(0.2)
    return job


def execute_sync(module_name, args=None, db=None, cfg=None):
    return execute(module_name, args, db, cfg, bg=False)


def parse_kv(pairs):
    out = {}
    for item in pairs:
        if "=" in item:
            k, v = item.split("=", 1)
            out[k] = v
    return out
```

```python
        self.cfg = Config(ws.name)
        self.module = None
        self.args = {}

    def prompt(self):
        tgt = self.args.get("target") or "-"
        name = self.module or "lw"
        return f"{col(name, M)}@{col(self.ws.name, Y)}:{col(tgt, C)}> "

    def loop(self):
        print(col("\n[*] console aktif. 'help' untuk perintah.\n", D))
        while True:
            try:
                line = input(self.prompt()).strip()
            except (EOFError, KeyboardInterrupt):
                print()
                return
            if not line:
                continue
            try:
                if self.dispatch(line):
                    return
            except KeyboardInterrupt:
                print(col("  dibatalkan", Y))
            except Exception as e:
                print(col(f"  [!] {e}", R))

    def dispatch(self, line):
        parts = line.split()
        cmd, rest = parts[0].lower(), parts[1:]

        if cmd in ("quit", "exit", "q"):
            return True
        if cmd == "help":
            print(HELP)
        elif cmd == "use":
            if not rest:
                print(col("  use <modul>", Y)); return
            meta = module_loader.get(rest[0])
            if not meta:
                cand = module_loader.search(rest[0])
                print(col(f"  tidak ada. mirip: {[c['name'] for c in cand[:6]]}", Y))
                return
            self.module = meta["name"]
            self.args = {"target": self.args.get("target")} if self.args.get("target") else {}
            print(col(f"  modul: {meta['name']} — {meta['desc']}", G))
            print(col(f"  {meta['source']}", D))
        elif cmd == "back":
            self.module = None
        elif cmd == "options":
            if not self.module:
                print(col("  belum ada modul", Y)); return
            meta = module_loader.get(self.module)
            for k, spec in (meta.get("params") or {}).items():
                cur = self.args.get(k) or spec.get("default") or self.cfg.get(k) or ""
                print(f"  {col(k, C):<22} {'WAJIB ' if spec.get('required') else ''}{cur}")
        elif cmd == "set":
            if len(rest) < 2:
                print(col("  set <k> <v>", Y)); return
            self.args[rest[0]] = " ".join(rest[1:])
        elif cmd == "unset":
            self.args.pop(rest[0] if rest else "", None)
        elif cmd == "target":
            if rest:
                self.args["target"] = rest[0]
                print(col(f"  target: {rest[0]}", G))
        elif cmd in ("run", "run!"):
            if not self.module:
                print(col("  pilih modul dulu", Y)); return
            runner.execute(self.module, self.args, self.ws.db, self.cfg, bg=(cmd == "run!"))
        elif cmd == "modules":
            cat = rest[0] if rest else None
            for c, label in module_loader.CATEGORIES.items():
                if cat and cat != c:
                    continue
                mods = module_loader.by_category(c)
                if not mods:
                    continue
                print(col(f"\n[{c}] {label}", B))
                for m in mods:
                    print(f"  {col(m['name'], C):<34} {D}{m['desc']}{W}")
        elif cmd == "search":
            for m in module_loader.search(" ".join(rest)):
                print(f"  {col(m['name'], C):<34} {D}{m['desc']}{W}")
        elif cmd == "info":
            m = module_loader.get(rest[0] if rest else "")
            if not m:
                print(col("  tidak ada", Y)); return
            print(f"  {m['name']} — {m['desc']}\n  kategori: {m['category']}\n"
                  f"  butuh: {', '.join(m.get('needs') or []) or '-'}\n"
                  f"  root: {'ya' if m.get('root') else 'tidak'}")
        elif cmd == "jobs":
            for j in runner.JOBS.all():
                print(f"  #{j.id} {j.status:<10} {j.module} {D}{time_since(j.started)}{W}")
        elif cmd == "kill":
            print(col("  dihentikan" if rest and runner.JOBS.kill(rest[0]) else "  id salah", Y))
        elif cmd == "sessions":
            for s in runner.SESSIONS.all():
                print(f"  #{s.id} {s.kind} {s.endpoint} pid={s.pid} {s.status}")
        elif cmd == "workspace":
            if not rest:
                print(col(f"  aktif: {self.ws.name}", G)); return
            from .workspace import set_current
            self.ws = Workspace(rest[0])
            self.cfg = Config(self.ws.name)
            set_current(rest[0])
            print(col(f"  pindah ke workspace: {rest[0]}", G))
        elif cmd == "findings":
            for f in self.ws.db.findings(60):
                print(f"  [{f['severity']}] {f['title']} — {f['target']} ({f['module']})")
        elif cmd == "creds":
            for c in self.ws.db.creds(60):
                print(f"  {c['target']} {c['username']}:{c['secret']}")
        elif cmd == "report":
            p = reporting.generate(self.ws.db, self.ws.name, rest[0] if rest else "md")
            print(col(f"  {p}", G))
        elif cmd == "exec":
            self.do_exec(rest)
        elif cmd == "doctor":
            for tool, ok, pkg in deps.doctor():
                print(f"  {'OK  ' if ok else 'MISS'} {tool:<22} {D}{pkg}{W}")
        else:
            print(col(f"  perintah tidak dikenal: {cmd}", R))
        return False

    def do_exec(self, rest):
        if not rest:
            print(col("  exec <bahasa> <file|->", Y)); return
        lang = rest[0]
        from . import exec_engine
        if lang not in exec_engine.LANGS:
            print(col(f"  bahasa: {', '.join(exec_engine.LANGS)}", Y)); return
        src = None
        if len(rest) > 1 and rest[1] != "-":
            src = open(rest[1]).read()
        else:
            print(col("  tempel kode, akhiri dengan baris 'EOF'", D))
            buf = []
            while True:
                l = input()
                if l.strip() == "EOF":
                    break
                buf.append(l)
            src = "\n".join(buf)
        r = exec_engine.run_source(lang, src, self.ws.dir / "code")
        print(r.text())


def time_since(t):
    import time
    d = int(time.time() - t)
    return f"{d}s" if d < 60 else f"{d // 60}m"
```

```python
"""Reconnaissance & footprinting."""
import re

from ..core import netutil as nu
from ._base import MODULES, mod

SUBS = ["www", "mail", "api", "dev", "staging", "test", "admin", "vpn", "git",
        "cdn", "app", "m", "blog", "shop", "portal", "db", "ftp", "smtp", "ns1", "ns2"]


@mod(name="recon.whois", category="recon", desc="Whois lookup domain/IP",
     needs=["whois"], params={"target": {"required": True, "desc": "domain / IP"}})
def whois(ctx):
    rc, out, err = ctx.exec(["whois", ctx.p("target")], 45)
    ctx.out(out or err)
    for pat, label in ((r"Registrar:\s*(.+)", "registrar"), (r"Creation Date:\s*(.+)", "created"),
                       (r"Registry Expiry Date:\s*(.+)", "expiry")):
        m = re.search(pat, out, re.I)
        if m:
            ctx.finding(f"whois {label}", "info", m.group(1).strip())
    ctx.artifact("whois.txt", out)
    return f"{len(out)} byte whois"


@mod(name="recon.dns", category="recon", desc="Enumerasi record DNS",
     params={"target": {"required": True, "desc": "domain"}})
def dns(ctx):
    t = ctx.p("target")
    for rtype in ("A", "AAAA", "MX", "TXT", "NS", "SOA", "CNAME"):
        if nu.have("dig"):
            rc, out, err = ctx.exec(["dig", "+short", rtype, t], 15)
        else:
            out = "\n".join(nu.resolve(t)) if rtype == "A" else ""
        out = out.strip()
        ctx.out(f"  {rtype:5} -> {out or '-'}")
        if out:
            ctx.finding(f"DNS {rtype}", "info", out[:400])
    return "dns selesai"


@mod(name="recon.revdns", category="recon", desc="Reverse DNS / PTR",
     params={"target": {"required": True, "desc": "IP"}})
def revdns(ctx):
    name = nu.reverse(ctx.p("target"))
    ctx.out(f"  PTR: {name or 'tidak ada'}")
    if name:
        ctx.finding("reverse DNS", "info", name)
    return name


@mod(name="recon.http_headers", category="recon", desc="Ambil header HTTP mentah",
     params={"target": {"required": True, "desc": "host atau URL"}})
def http_headers(ctx):
    t = ctx.p("target")
    if not t.startswith("http"):
        t = "https://" + t
    status, headers, body = nu.http_get(t, ctx.pi("timeout"))
    ctx.out(f"  status {status}")
    for k, v in headers.items():
        ctx.out(f"  {k}: {v}")
    ctx.finding("header HTTP", "info", f"status {status}", "\n".join(f"{k}: {v}" for k, v in headers.items()))
    return f"status {status}"


@mod(name="recon.tech_detect", category="recon", desc="Deteksi teknologi web",
     params={"target": {"required": True, "desc": "host atau URL"}})
def tech_detect(ctx):
    t = ctx.p("target")
    if not t.startswith("http"):
        t = "http://" + t
```

```python
"""Vulnerability assessment."""
import json
import re

from ..core import netutil as nu
from ._base import MODULES, mod

NVD = "https://services.nvd.nist.gov/rest/json/cves/2.0"


def _nvd(ctx, query):
    status, _, body = nu.http_get(f"{NVD}?{query}&resultsPerPage=20", 45)
    try:
        data = json.loads(body.decode(errors="ignore"))
    except Exception:
        ctx.out("  gagal baca NVD"); return []
    rows = []
    for item in data.get("vulnerabilities", []):
        c = item["cve"]
        score = 0.0
        for key in ("cvssMetricV31", "cvssMetricV30", "cvssMetricV2"):
            if c.get("metrics", {}).get(key):
                score = float(c["metrics"][key][0]["cvssData"]["baseScore"]); break
        desc = next((d["value"] for d in c["descriptions"] if d["lang"] == "en"), "")
        rows.append((c["id"], score, desc))
    return rows


def _emit(ctx, rows):
    for cid, score, desc in rows:
        sev = "critical" if score >= 9 else "high" if score >= 7 else "medium" if score >= 4 else "low"
        ctx.out(f"  {cid} score={score} {desc[:150]}")
        ctx.finding(cid, sev, desc[:300], f"CVSS {score}")
    return f"{len(rows)} CVE"
```

```python
     params={"target": {"required": True, "desc": "path file"},
             "pattern": {"default": "pass|key|token|http|flag|admin|secret"}})
def strings(ctx):
    rc, out, err = ctx.exec(["strings", "-n", "5", ctx.p("target")], 120)
    hits = [l for l in out.splitlines() if re.search(ctx.p("pattern"), l, re.I)]
    for h in hits[:200]:
        ctx.out(f"  {h}")
    if hits:
        ctx.finding("string sensitif", "medium", f"{len(hits)} kandidat", "\n".join(hits[:100]))
    return f"{len(hits)} hit"


@mod(name="bin.elf", category="binary", desc="Header ELF + section", needs=["readelf"],
     params={"target": {"required": True, "desc": "path ELF"}})
def elf(ctx):
    for args in (["-hW"], ["-lW"], ["-SW"]):
        rc, out, err = ctx.exec(["readelf"] + args + [ctx.p("target")], 30)
        ctx.out(out[:2500] or err)
    return "elf selesai"


@mod(name="bin.checksec", category="binary", desc="Cek hardening binary (NX/PIE/RELRO)", needs=["readelf"],
     params={"target": {"required": True, "desc": "path ELF"}})
def checksec(ctx):
    rc, hdr, _ = ctx.exec(["readelf", "-hW", ctx.p("target")], 20)
    rc2, prog, _ = ctx.exec(["readelf", "-lW", ctx.p("target")], 20)
    rc3, dyn, _ = ctx.exec(["readelf", "-dW", ctx.p("target")], 20)
    pie = "DYN" in hdr.split("Type:")[1][:40] if "Type:" in hdr else False
    nx = "GNU_STACK" in prog and "RWE" not in re.search(r"GNU_STACK.*", prog).group(0)
    relro = "GNU_RELRO" in prog
    now = "BIND_NOW" in dyn
    ctx.out(f"  PIE: {'ya' if pie else 'TIDAK'}  NX: {'ya' if nx else 'TIDAK'}  "
            f"RELRO: {'full' if relro and now else 'partial' if relro else 'TIDAK'}")
    for name, bad, sev in (("tanpa PIE", not pie, "medium"), ("stack executable", not nx, "high"),
                           ("tanpa RELRO", not relro, "low")):
        if bad:
            ctx.finding(name, sev, ctx.p("target"))
    return "checksec selesai"


@mod(name="bin.disasm", category="binary", desc="Disassembly fungsi", needs=["objdump"],
     params={"target": {"required": True, "desc": "path ELF"},
             "symbol": {"default": "main", "desc": "nama fungsi"}})
def disasm(ctx):
    rc, out, err = ctx.exec(["objdump", "-d", "-M", "intel",
                             f"--disassemble={ctx.p('symbol')}", ctx.p("target")], 60)
    ctx.out(out[:5000] or err)
    ctx.artifact("disasm.asm", out)
    return "disasm selesai"


@mod(name="bin.rop", category="binary", desc="Cari gadget ROP", needs=["ROPgadget"],
     params={"target": {"required": True, "desc": "path ELF"},
             "pattern": {"default": "pop rdi"}})
def rop(ctx):
    rc, out, err = ctx.exec(["ROPgadget", "--binary", ctx.p("target"),
                             "--only", "pop|ret"], 180)
    lines = [l for l in out.splitlines() if ctx.p("pattern") in l]
    for l in lines[:60]:
        ctx.out(f"  {l}")
    ctx.artifact("gadgets.txt", "\n".join(lines))
    return f"{len(lines)} gadget"


@mod(name="bin.binwalk", category="binary", desc="Binwalk signature scan", needs=["binwalk"],
     params={"target": {"required": True, "desc": "path file"}})
def binwalk(ctx):
    rc, out, err = ctx.exec(["binwalk", ctx.p("target")], 180)
    ctx.out(out or err)
    return "binwalk selesai"


@mod(name="bin.frida", category="binary", desc="Proses frida-server aktif", offline=True)
def frida(ctx):
    rc, out, err = ctx.exec(["frida-ps", "-U"], 40)
    if rc != 0:
        ctx.out("  frida-ps tidak ada / server belum jalan")
        return None
    ctx.out(out[:3000])
    ctx.finding("frida terhubung", "info", f"{len(out.splitlines())} proses")
    return "frida selesai"


@mod(name="bin.hash", category="binary", desc="Hash file (sha256/md5)", offline=True,
     params={"target": {"required": True, "desc": "path file"}})
def hashfile(ctx):
    import hashlib
    h256 = hashlib.sha256()
    hmd5 = hashlib.md5()
    with open(ctx.p("target"), "rb") as fh:
        for chunk in iter(lambda: fh.read(1 << 16), b""):
            h256.update(chunk); hmd5.update(chunk)
    ctx.out(f"  sha256 {h256.hexdigest()}\n  md5    {hmd5.hexdigest()}")
    ctx.finding("hash file", "info", h256.hexdigest())
    return h256.hexdigest()
```

```bash
OUT="lonewolf-2026-$(date +%Y%m%d-%H%M).zip"

skip = {"__pycache__", ".git", ".lw"}
with zipfile.ZipFile(out, "w", zipfile.ZIP_DEFLATED) as z:

        if p.is_file():
            z.write(p, p.as_posix())
print("dibuat:", out)
PY
sha256sum "$OUT" > "$OUT.sha256" 2>/dev/null || true
echo "[+] $OUT"
```
